cmake_minimum_required(VERSION 3.22)
project(CleanPiano VERSION 1.0.0)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)
FetchContent_Declare(JUCE
    GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    GIT_TAG 8.0.4
    GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)

juce_add_plugin(CleanPiano
    COMPANY_NAME "CleanAudio"
    IS_SYNTH TRUE
    NEEDS_MIDI_INPUT TRUE
    PLUGIN_MANUFACTURER_CODE Clna
    PLUGIN_CODE Cpno
    FORMATS VST3 Standalone
    PRODUCT_NAME "Clean Piano"
    COPY_PLUGIN_AFTER_BUILD FALSE)

target_sources(CleanPiano PRIVATE Source/Plugin.cpp)

target_compile_definitions(CleanPiano PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0)

target_link_libraries(CleanPiano PRIVATE
    juce::juce_audio_utils
    juce::juce_recommended_config_flags
    juce::juce_recommended_lto_flags)
